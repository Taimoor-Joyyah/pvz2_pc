function(runtime_target target)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_include_directories(${target} PUBLIC "${PROJECT_SOURCE_DIR}/include")
  target_compile_definitions(${target} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  if(MSVC)
    target_compile_options(${target} PRIVATE /Zc:__cplusplus /permissive- /W4 /EHsc /utf-8)
    set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    if(ACP_PROFILE_SYMBOLS)
      target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:/Zi>")
    endif()
    get_target_property(target_kind ${target} TYPE)
    if(target_kind STREQUAL "EXECUTABLE")
      target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:/OPT:REF>" "$<$<CONFIG:Release>:/OPT:ICF>" "$<$<CONFIG:Release>:/INCREMENTAL:NO>")
      if(ACP_PROFILE_SYMBOLS)
        target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:/DEBUG:FULL>")
      endif()
    endif()
  endif()
endfunction()
